// ─── iMAX FUNDED — Rules Engine ──────────────────────────────────────────────
// Drop into Node.js cron, Supabase Edge Function, or call after each trade sync.
// Runs after every MT5 trade is written to your DB.

const RULES = {
  dailyLossLimit:  0.05,   // 5%  of starting balance
  maxDrawdown:     0.10,   // 10% from starting balance (ever)
  profitTarget:    0.10,   // 10% profit to pass
  minTradingDays:  5,      // at least 5 calendar days with a closed trade
};

function evaluateAccount(account, trades) {
  const { startingBalance } = account;
  const today = new Date().toISOString().slice(0, 10);

  // ── 1. Daily P&L ─────────────────────────────────────────────────────────
  const todayTrades = trades.filter(t => t.closeTime.startsWith(today));
  const dailyPnl    = todayTrades.reduce((s, t) => s + t.pnl, 0);
  const dailyLoss   = Math.min(0, dailyPnl);   // negative or 0
  const dailyPct    = Math.abs(dailyLoss) / startingBalance;

  // ── 2. Total P&L & running balance ────────────────────────────────────────
  const totalPnl     = trades.reduce((s, t) => s + t.pnl, 0);
  const balance      = startingBalance + totalPnl;
  const profitPct    = totalPnl / startingBalance;

  // ── 3. Max drawdown (from peak equity) ────────────────────────────────────
  let peak = startingBalance, maxDD = 0, running = startingBalance;
  for (const t of trades) {
    running += t.pnl;
    if (running > peak) peak = running;
    const dd = (peak - running) / startingBalance;
    if (dd > maxDD) maxDD = dd;
  }

  // ── 4. Minimum trading days ───────────────────────────────────────────────
  const tradingDays = new Set(trades.map(t => t.closeTime.slice(0,10))).size;

  // ── 5. Evaluate ───────────────────────────────────────────────────────────
  const breaches = [];
  if (dailyPct  >= RULES.dailyLossLimit) breaches.push('DAILY_LOSS_LIMIT');
  if (maxDD      >= RULES.maxDrawdown)   breaches.push('MAX_DRAWDOWN');

  const passed =
    breaches.length === 0
    && profitPct   >= RULES.profitTarget
    && tradingDays >= RULES.minTradingDays;

  return { balance, dailyPnl, dailyPct, profitPct, maxDD, tradingDays, breaches, passed };
}

export default evaluateAccount;
